FILE Toolbox\Analyses\Deterministic Analysis\classify_Attractors.m
function bool = classify_Attractors(obj)
try
    cellFateDeterminationLogic = obj.CellFateDeterminationLogic;
    uniqueAttractors = obj.AttractorTable(:,1);
    partialAttractorList = cellfun(@(a) a(:, obj.OutputNodesIndices),...
        uniqueAttractors, 'un', 0);
    upperbound = 0.2;
    lowerbound = 0.2;
    [attractorSummaryInFraction, attractorSummaryAsWhole] = ...
        cellfun(@(a)summarize_AttractorActivity (a, upperbound, ...
        lowerbound), partialAttractorList, 'un', 0);
    associatedCellFates = cellfun(@(attr) associate_CellFates...
        (cellFateDeterminationLogic, attr,  numel(obj.UniqueCellFates)), ...
        attractorSummaryAsWhole);
    attractorTable = obj.AttractorTable(:, 1:3);
    attractorTable = horzcat(attractorTable, associatedCellFates);
    [uniqueAssociatedCellFates , ~, ~] = unique(associatedCellFates);
    fatePropensities = cell(numel(uniqueAssociatedCellFates), 1);
    for i = 1:numel(fatePropensities)
        fatePropensities{i}  = sum([attractorTable{ismember...
            (associatedCellFates, uniqueAssociatedCellFates(i)), 3}], 2);
    end
    cellFateLandscape        = cell(numel(uniqueAssociatedCellFates), 2);
    cellFateLandscape(:,1)   = uniqueAssociatedCellFates;
    cellFateLandscape(:,2)   = fatePropensities;
    obj.AttractorSummaryInFraction = attractorSummaryInFraction;
    obj.AttractorSummaryAsWhole    = attractorSummaryAsWhole;
    obj.AttractorTable             = attractorTable;
    obj.CellFateLandscape          = flipud(sortCellArr(cellFateLandscape, 2));
    bool = 1;
catch
    bool = 0;
end
end

        uiwait(errordlg('The selected mat file is either empty or contains multiple variables. Please input mat file containing only one network state list.', 'Invalid mat file'));
        enableDisableFig(findall(0, 'tag','ATLANTIS_Main'), 'off');

        uiwait(errordlg('The selected csv file is empty. Please input valid csv. Press help for more information.', 'Invalid mat file'));
        enableDisableFig(findall(0, 'tag','ATLANTIS_Main'), 'off');
        return;
    end

        try
            stateCombinations = fileInfo{2,:, :};
        catch
            stateCombinations = [];
        end
    else
        stateCombinations = fileInfo;
    end 
end
try
    if isempty(stateCombinations)
        undefined/0;
    end
    isboolean = all((stateCombinations == 1) == (stateCombinations ~= 0) == 1);
    if ~isboolean
        undefined/0;
    end
    clear isboolean;
    if ~(size(stateCombinations, 2) == nodeCount)
        undefined/0;
    end

FILE Toolbox\Analyses\Deterministic Analysis\perform_DeterministicAnalysis.m
function [bool, message] = perform_DeterministicAnalysis(myNetwork, totalProgress)
message = '';
tic;
networkType = myNetwork.NetworkType;
if strcmp(networkType, 'WeightsBased')
    bool = deterministicEngine_WeightsBased(myNetwork, totalProgress);
else
    bool = deterministicEngine_RulesBased(myNetwork, totalProgress);
end
if isempty(myNetwork.AttractorTable) 
    message = 'Could not find any attractors. Possible reasons might be; 1) Small attractor search duration value or 2) The input network is not robust and hence does not achieve a stable state. Please retry after increasing the attractor search duration value.';
    bool = 0;
    return;
end
if bool
    if ~isempty(myNetwork.CellFateDeterminationLogic)
        bool = classify_Attractors(myNetwork);
        if ~bool            
            message = 'Something went wrong while characterizing attractors. Please reinput cell fate logic or reperform deterministic analysis.';
            return;
        end
    end
else
    message = 'Something went wrong. Please reperform deterministic analysis. If this does not work, restart ATLANTIS. If the problem persists please report this issue.';
    return;
end
disp(['total time for DA and cell fate association: ', num2str(toc)]);
end

FILE Toolbox\Analyses\Probabilistic Analysis\perform_ProbabilisticAnalysis.m
function [bool, message] = perform_ProbabilisticAnalysis(myNetwork, totalProgress)
tic;
message = '';
networkType = myNetwork.NetworkType;
if strcmp(networkType, 'WeightsBased')
    bool = find_StateTransitionProbabilities(myNetwork, totalProgress);
    disp(['Time Taken to Find State Transitions Probabilities: ', num2str(toc)]);
    if ~bool
        message = 'Error while finding state transition probabilities (Possible erors; 1) Your system memory is insufficient. Please reduce the network state list size and try again)';
        bool = 0;
        return;
    end
    bool = find_SteadyStateProbabilities(myNetwork, totalProgress);
    disp(['Time Taken to Find Steady State Probabilites: ', num2str(toc)]);
    if ~bool
        message = 'Error while finding steady state probabilities';
        bool = 0;
        return;
    end

function associatedFates  = associate_CellFates(cellFateInfo, attr, numOfUnCF)
cellfates = cellFateInfo{:,1};
logicList = cellFateInfo{:,2};
associatedFates = cell(numOfUnCF, 1);
fateIndex = 1;
replacingNaNs = 1000;
for i = 1:numel(logicList)
    tempAttr = attr;
    logic = logicList{i};
    allNaNs = isnan(logic);
    logic(allNaNs) = replacingNaNs;
    tempAttr(allNaNs) = replacingNaNs;
    if isequal(logic, tempAttr)
        associatedFates{fateIndex} = cellfates{i};
        fateIndex = fateIndex + 1;
    end
end
emptyCells = cellfun(@isempty, associatedFates);
associatedFates(emptyCells) = [];
[associatedFates, ~, ~] = uniqueCellArray(associatedFates);
associatedFates = { joinStr_CellArr(associatedFates, ' + ' ) };
end

function states = generate_StateCombinations_randomSampling(myNetwork, sampleSize)
states = []; % will store randomly generated states
progressbar('Generating States. Please Wait...');
while size(states, 1) < sampleSize
    i = size(states, 1);
    diff = sampleSize - size(states, 1);
    progressbar(i/sampleSize);
    states = vertcat(states, randi([0 1], diff,myNetwork.NodeCount)); %#ok<AGROW>
    [a, b]=unique(states,'rows');   %#ok<ASGLU> % finding replicated rows %
    states = states(sort(b),:);     % removing replicated rows
end
progressbar(1);
end

end

FILE Toolbox\Analyses\Supplementary functions\parse_CellFateLogic 2.m
function [bool, rawLogic] = parse_CellFateLogic(myNetwork, fileName, pathName)
rawLogic = {};
try
    allData = csv2cell(fullfile(pathName, fileName));
    if isempty(allData)
       undefined/0; 
    end
    cellfates = allData(:,1);
    cellfates = cellfates(2:end);
    areNum = cellfun(@str2num, cellfates, 'un', 0);
    empty = cellfun(@isempty, areNum);
    if all(~empty)
        undefined/0;
    end
    outputNodes = allData(1, :);
    cellFateDeterminationLogic = outputNodes;
    outputNodes = outputNodes(2:end);
    [~, outputNodeIndicies] = ismember(cellfun(@lower, outputNodes, 'un', 0), cellfun(@lower, myNetwork.NodeNames, 'un', 0));
    if sum(outputNodeIndicies == 0) > 0
        uiwait(errordlg('Some of the nodes in this file are not present in the network. Please only input file containing nodes same as that in the network.', 'Invalid File'));
        undefined/0;
    end
    row = 2;
    nodePosisitions = [];
    for f = 1:numel(cellfates)
        cellFateDeterminationLogic{row, 1} = cellfates{f};
        listOfNodestates = cell(1, numel(outputNodes));
        for n = 1:numel(outputNodes)
            Nodestates = eval(strcat('[' ,allData{f+1, n+1} ,']'));
            if ~isnumeric(Nodestates)
                undefined/0;
            end
            if ~isempty(Nodestates)
                nodePosisitions = horzcat(nodePosisitions, n);
            end
            listOfNodestates{n} = Nodestates;
        end
        listOfNodestates(cellfun(@isempty, listOfNodestates)) = [];
        combinations = generateCombinations(listOfNodestates{:});
        rowToBeCopied = cellFateDeterminationLogic(row, :);
        newRows = repmat(rowToBeCopied, [size(combinations, 1)-1, 1]);
        cellFateDeterminationLogic = vertcat(cellFateDeterminationLogic, newRows);
        lowerBound = row;
        upperBound = row + size(combinations, 1) - 1;
        cellFateDeterminationLogic(lowerBound:upperBound, nodePosisitions+1) = num2cell(combinations);
        row = row + size(combinations, 1);
        nodePosisitions = [];
    end
    rawLogic = cellFateDeterminationLogic;
    rules = cellFateDeterminationLogic(2:end, 2:end);
    rules(cellfun(@isempty, rules)) = {NaN};
    rules = num2cell(cell2mat(rules), 2);
    cellFateDeterminationLogic = {cellFateDeterminationLogic(2:end, 1), rules};
    bool = 1;
    myNetwork.CellFateDeterminationLogic = cellFateDeterminationLogic;
    myNetwork.UniqueCellFates = cellfates;
    myNetwork.OutputNodes = outputNodes;

                endMark = step;
                attractor = trajectoryVec(startMark:endMark, :);

            end

        end

    function attractor = searchForAttractor(currentState, targetNodeIndices, sourceNodeIndices, updateLogics, progressbarInfo)
        attractor = []; 
        searchDuration = myNetwork.AttractorSearchDuration; 
        trajectoryCell = cell(1,searchDuration);
        trajectoryVec = zeros(searchDuration, myNetwork.NodeCount);
        for step = 1:searchDuration
            trajectoryCell{step} = currentState;
            trajectoryVec(step, :) = currentState;
            nextState = UpdateState(currentState);
            index = find(cellfun(@(a) isequal(a,nextState), trajectoryCell(1:step)), 1); % check if updated state is already present in the trajectory
            if ~isempty(index) % IF TRUE THAN CYCLE FOUND
                trajectoryVec(step+1:end, :) = [];
                startMark = size(trajectoryVec, 1) - (step - index);
                endMark = step;
                attractor = trajectoryVec(startMark:endMark, :);
                break;
            end
            currentState = nextState;
        end
        function nextstate = UpdateState(currentState)
            nextstate = currentState;
            for i = 1:size(sourceNodeIndices,1)
                nextstate(i) = UpdateNodeActivty(sourceNodeIndices{i}, targetNodeIndices{i});
            end
            function updatedNode = UpdateNodeActivty(SourceNodeIndex, targetNodeIndex)
                updatedNode = updateLogics{targetNodeIndex}(binary2Decimal(currentState(SourceNodeIndex)) + 1);
                if isempty(updatedNode)
                    updatedNode = currentState(targetNodeIndex);
                end
            end
        end
    end

function bool = find_StateTransitionProbabilities(myNetwork, totalProgress)
try    
    cancelButton = CancelSimulationButton;
    movegui(cancelButton, 'northeast');
    setappdata(0,'CancelSimulation', []);
    if myNetwork.PruneStateSpace == 1;
        prunnedStateSpace = StateSpacePruning(myNetwork);
        myNetwork.PrunnedStateSpacePA = prunnedStateSpace;
        initialStates = prunnedStateSpace;
    else
        initialStates = myNetwork.NetworkStateList;
    end
    numberOfStates = size(initialStates, 1);
    aij = myNetwork.EdgeWeights;
    noise = myNetwork.Noise;
    c = myNetwork.DegradationConstant;
    bi = myNetwork.BasalValues';
    Tij = ones(numberOfStates, numberOfStates);
    nodeOnProbabilities = zeros(numberOfStates, myNetwork.NodeCount);
    Summation = initialStates * aij + repmat(bi,[numberOfStates,1]);
    indices = Summation > 0;
    nodeOnProbabilities(indices) = 1/2 + 1/2*tanh(noise*Summation(indices));
    indices = Summation < 0;
    nodeOnProbabilities(indices) = 1 - (1/2 - 1/2*tanh(noise*Summation(indices)));
    indices = find(Summation == 0);
    for n = 1:length(indices)
        if initialStates(indices(n)) == 0
            nodeOnProbabilities(indices(n)) = c;
        elseif initialStates(indices(n)) == 1
            nodeOnProbabilities(indices(n)) = 1-c;
        end
    end
    activeNodes = initialStates == 0;
    inactiveNodes   = initialStates == 1;
    for Statei = 1:numberOfStates
        progressbar(totalProgress, Statei/numberOfStates, 0);
        allOn = repmat(nodeOnProbabilities(Statei,:), [numberOfStates, 1]);
        allOff = 1 - allOn;
        allOn(activeNodes) = 1;
        allOff(inactiveNodes) = 1;
        Tij(Statei, :) = (prod(allOn, 2) .* prod(allOff, 2));
        if ~isempty(getappdata(0,'CancelSimulation'))
            cancelButton.delete;
            undefined/0;
        end
    end
    cancelButton.delete;
    myNetwork.StateTransitionProbabilities = Tij;
    bool = 1;
catch
    bool = 0;
    progressbar(totalProgress, 1, 0);
    cancelButton.delete;
end
end

function bool = find_SteadyStateProbabilities(myNetwork, totalProgress)
try
    cancelButton = CancelSimulationButton;
    movegui(cancelButton, 'northeast');
    setappdata(0,'CancelSimulation', []);
    if myNetwork.PruneStateSpace == 1;
        stateSpaceSize = size(myNetwork.PrunnedStateSpacePA, 1);
    else
        stateSpaceSize = size(myNetwork.NetworkStateList, 1);
    end
    maxSteps = myNetwork.TimeSteps;
    Tij = myNetwork.StateTransitionProbabilities;
    Tji = Tij';
    initialProbabilities = ones(stateSpaceSize, 1)./(stateSpaceSize);
    iteration = 1;
    errorThreshold = myNetwork.ErrorThreshold;
    while (iteration <= maxSteps)
        fluxOut = zeros(stateSpaceSize, 1);
        fluxIn = zeros(stateSpaceSize, 1);
        progressbar(totalProgress, 1, iteration/maxSteps);
        for Statei = 1:stateSpaceSize
            fluxOut = Tij(:,Statei).* initialProbabilities...
                + fluxOut;
            fluxIn = Tji(:,Statei).* initialProbabilities(Statei)...
                + fluxIn;
        end
        changeInProbabilities = - fluxOut + fluxIn;
        updatedProbabilites = initialProbabilities + changeInProbabilities;
        diff = abs(initialProbabilities - updatedProbabilites);
        PercentageSimilar = sum( diff <= errorThreshold , 1 ) / stateSpaceSize;
        if PercentageSimilar >= 0.98 && iteration > 10 && all(initialProbabilities >= 0)
            disp(['Steady State Probabilities reached at iteration #: ', num2str(iteration)]);
            myNetwork.SteadyStateTimeStep = iteration;
            break;
        end
        initialProbabilities = updatedProbabilites;
        iteration = iteration + 1;
        if ~isempty(getappdata(0,'CancelSimulation'))
            cancelButton.delete;
            undefined/0;
        end
    end
    cancelButton.delete;
    myNetwork.SteadyStateProbabilities = updatedProbabilites;
    bool = 1;
catch
    bool = 0;
    progressbar(totalProgress, 1, 1);
    cancelButton.delete;
end
end

        outputNodeStates = cellfun(@(a) a(:, myNetwork.OutputNodesIndices), ...
            num2cell(transientStates(:, 2:end), 2), 'un', 0);
        associatedCellFates = cellfun(@(attr) associate_CellFates...
            (myNetwork.CellFateDeterminationLogic, attr, ...
            numel(myNetwork.UniqueCellFates)), outputNodeStates);
    else
        associatedCellFates = cell(size(transientStates, 1), 1);
    end
    myNetwork.Trajectory = {indicies, associatedCellFates, transientStates};
    bool = 1;
catch
    bool = 0;
end
end

function bool = parse_TrajectoryBounds(obj, fileName, pathName)
try
    allData = importdata(fullfile(pathName, fileName));
    if isempty(allData) || isstruct(allData)
       undefined/0; 
    end
    if ~ismatrix(allData)
        undefined/0;
    end
    if size(allData, 1) ~= 2 || size(allData, 2) ~= obj.NodeCount
        undefined/0;
    end
    obj.TrajectoryBounds = allData;
    bool = 1;
catch
    bool = 0;
end
end

FILE Toolbox\Analyses\Probabilistic Analysis\Supplemetary functions\save_ProbabilisticAnalysisResults.m
function bool = save_ProbabilisticAnalysisResults( myNetwork )
try
    progressbar('Saving results...');
    if ~isempty(myNetwork.FileName)
        PathName = [myNetwork.ResultFolderName, strrep(myNetwork.FileName, ' - ', filesep), 'Probabilistic Analysis Output', filesep];
    else
        PathName = [myNetwork.ResultFolderName, 'Probabilistic Analysis Output', filesep];
    end
    [~, ~] = mkdir(PathName);
    if myNetwork.PruneStateSpace == 1
        stateSpace = myNetwork.PrunnedStateSpacePA;
    else
        stateSpace = myNetwork.NetworkStateList;
    end
    if ~isempty(myNetwork.SteadyStateProbabilities)
        FileName2 = [myNetwork.FileName, 'Steady State Probabilities', '.csv'];
        names = myNetwork.NodeNames;
        if size(names, 1) > 1
            names = names';
        end
        SSPs = myNetwork.SteadyStateProbabilities;
        progressbar(0.2/1);
        if ~isempty(myNetwork.CellFateDeterminationLogic)
            associatedCellFates = myNetwork.AssociatedCellFates;
            if size(associatedCellFates, 1) ~= size(stateSpace, 1)
                [~, locA] = ismember(stateSpace, myNetwork.NetworkStateList, 'rows');
                if min(locA) == 0
                    outputNodeActivities = cellfun(@(a) a(:, myNetwork.OutputNodesIndices), ...
                        num2cell(stateSpace, 2), 'un', 0);
                    associatedCellFates = cellfun(@(attr) associate_CellFates...
                        (myNetwork.CellFateDeterminationLogic, attr, ...
                        numel(myNetwork.UniqueCellFates)), outputNodeActivities);
                else
                    associatedCellFates = associatedCellFates(locA);
                end
            end
        else
            associatedCellFates = cell(size(stateSpace, 1), 1);
        end
        header = horzcat({'Attractor #'}, 'Associated Cell Fate', 'Steady State Probability');
        try
            fid = fopen(strcat(PathName, FileName2), 'w');
        catch
            fid = fopen(strcat(PathName, FileName2,' (2)'), 'w');
        end
        formatToken = '%s,';
        fprintf(fid, formatToken, header{:, 1:3}, names{:});
        fprintf(fid, ['\', 'n']);
        for i = 1:size(SSPs, 1)  
            attrNumb = int2str(i);
            associatedFate = associatedCellFates{i};
            propensity = num2str(SSPs(i));
            fprintf(fid, '%s,', attrNumb, associatedFate, propensity);
            fprintf(fid, '%u,', stateSpace(i, :));
            fprintf(fid, ['\', 'n']);
        end
        progressbar(0.4/1);
        fclose(fid);
        clear attrCount associatedCellFates SSPs
    end
    if ~isempty(myNetwork.Trajectory)
        FileName = [myNetwork.FileName, 'Mapped Trajectory', '.csv'];
        trajectory = myNetwork.Trajectory;
        statesInTrajectory = trajectory{3}(:, 2:end);
        stateProbabilites = trajectory{3}(:, 1);
        stateNum = trajectory{1};
        associatedCellFates = trajectory{2};
        names = myNetwork.NodeNames;
        if size(names, 1) > 1
            names = names';
        end
        progressbar(0.6/1);
        header = horzcat({'#'}, 'Associated Cell Fate', 'Probabilities', names);
        try
            fid = fopen(strcat(PathName, FileName), 'w');
        catch
            fid = fopen(strcat(PathName, FileName,' (2)'), 'w'); 
        end
        formatToken = '%s,';
        fprintf(fid, formatToken, header{:, 1:3}, names{:});
        fprintf(fid, ['\', 'n']);
        for i = 1:size(stateNum, 1)  
            attrNumb = int2str(stateNum(i));
            associatedFate = associatedCellFates{i};
            propensity = num2str(stateProbabilites(i));
            fprintf(fid, '%s,', attrNumb, associatedFate, propensity);
            fprintf(fid, '%u,', statesInTrajectory(i, :));
            fprintf(fid, ['\', 'n']);
        end
        fclose(fid);
        progressbar(0.8/1);
    end
    if ~isempty(myNetwork.SteadyStateTimeStep) && myNetwork.SteadyStateTimeStep > 0
        FileName = [myNetwork.FileName, ' Steady State iteration'];
        fid6 = fopen(strcat(PathName, FileName, '.txt'), 'w');        
        fprintf(fid6, '%s', ['Iterations (Simulation Timesteps) taken to find steady state probabilities: ', num2str(myNetwork.SteadyStateTimeStep)]);
        fclose(fid6);
    end
    clear data header stateProbabilites statesInTrajectory trajectory associatedCellFates
    if myNetwork.PruneStateSpace == 1        
        FileName = 'Prunned Network State List PA';
    else
        FileName = 'Network State List PA';
    end
    csvwrite(strcat(PathName, FileName, '.csv'), stateSpace);
    save(strcat(PathName, FileName, '.mat'), 'stateSpace', '-v7.3');
    progressbar(1/1);
    bool = 1;
catch
    bool = 0;
    progressbar(1/1);
    fclose all;
end
end

FILE Toolbox\Analyses\Probabilistic Analysis\Supplemetary functions\StateSpacePruning.m
function prunnedStateSpace = StateSpacePruning(myNetwork)
initialStatesList = myNetwork.NetworkStateList;
updatedStateList = initialStatesList;
prunnedStateSpace = [];
if size(myNetwork.BasalValues, 1) > 1
    bi = myNetwork.BasalValues';
else
    bi = myNetwork.BasalValues;
end
NodeStateUpdateSummmation = initialStatesList*(myNetwork.EdgeWeights) + repmat(bi, [size(initialStatesList, 1), 1]);
indices = NodeStateUpdateSummmation > 0;
updatedStateList(indices) = 1;
indices = NodeStateUpdateSummmation < 0;
updatedStateList(indices) = 0;
indices = find(NodeStateUpdateSummmation == 0);
updatedStateList(indices) = initialStatesList(indices);
prunnedStateSpace = unique([prunnedStateSpace; updatedStateList], 'rows');
end
