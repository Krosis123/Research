function SammonProjection_Callback(hObject, eventdata, handles)
myNetwork = get(0, 'UserData');
if 2^(myNetwork.NodeCount) < 2^7
    uiwait(errordlg('Sammon mapping does not work well for state space smaller than 2^7', ''));
    set(hObject, 'Value', 0);
    set(handles.NaiveProjection, 'Value', 1);
else
    len = size(myNetwork.NetworkStateList, 1);
    if (len ~= 2^7 && len ~= 2^8 && len ~= 2^9 && len ~= 2^10 && len ~= 2^11 && len ~= 2^12) || len > 2^12
        len1 = size(myNetwork.ModifiedStateSpace, 1);
        if (len1 ~= 2^7 && len1 ~= 2^8 && len1 ~= 2^9 && len1 ~= 2^10 && len1 ~= 2^11 && len1 ~= 2^12) || len1 > 2^12
            uiwait(errordlg('Sammon mapping requires the state space to be 2^x, where x can be from 7-12. Current state space is not suitable to be projected to 2-Dimensions using sammon mapping. Size of state space needs to be modified. ', 'Warning'));
            uiwait(StateSpaceSizeSelectionMenu);
            userChoice = getappdata(0, 'StateSpaceOption');
            if userChoice == 0 % CANCEL
                set(hObject, 'Value', 0);
                set(handles.NaiveProjection, 'Value', 1);
                return;
            elseif userChoice == 1 % OPTIMAL
                possibleSizes = [2^7, 2^8, 2^9, 2^10, 2^11, 2^12];
                [~, mostSimilar] = min(abs(possibleSizes - size(len, 1)));
                newStateSpace = generate_StateCombinations_randomSampling(myNetwork, possibleSizes(mostSimilar));
            else % USER SELECTED
                newStateSpace = generate_StateCombinations_randomSampling(myNetwork, str2num(userChoice));
            end
            myNetwork.ModifiedStateSpace = newStateSpace;
        end
    end
end

function [prominentSteadyStateIndicies, transientStateindicies, attractorIndicies, attractorStateMarks, transientStateMarks, prominentSteadyStateMarks, arrowsObj] = annotate_AttractorLandscape(x, y, zData, stateSpaceLen, mappingMethod, randomInd, attractorIndiciesOriginal, transientStateindiciesOriginal, prominentSteadyStateIndiciesOriginal )
if strcmp(mappingMethod, 'sammon')
    z = zData{1};
    Z = zData{2};
    precision = length(num2str(10*(x(1)-floor(Z(1)))))/4;
    tol = 10^-(precision);
    shiftBy = 0.2*stateSpaceLen/100;
    [~, attractorIndicies] = ismembertol(z(attractorIndiciesOriginal), Z(:), tol, 'ByRows', true);
    [~, transientStateindicies] = ismembertol(z(transientStateindiciesOriginal), Z(:), tol, 'ByRows', true);
    [~, prominentSteadyStateIndicies] = ismembertol(z(prominentSteadyStateIndiciesOriginal), Z(:), tol, 'ByRows', true);
elseif strcmp(mappingMethod, 'naive')
    Z = zData;
    shiftBy = 1.2*stateSpaceLen/100;
    [~, attractorIndicies] = ismember(attractorIndiciesOriginal, randomInd);
    [~, transientStateindicies] = ismember(transientStateindiciesOriginal, randomInd);
    [~, prominentSteadyStateIndicies] = ismember(prominentSteadyStateIndiciesOriginal, randomInd);
end
attractorIndicies(attractorIndicies == 0) = [];
transientStateindicies(transientStateindicies == 0) = [];
prominentSteadyStateIndicies(prominentSteadyStateIndicies == 0) = [];
hold on;
zShift = 0.005;
infac = 0.02;
if ~isempty(attractorIndiciesOriginal)        
    markerColor =  [0.8216 0.2176 0.8137]; % ATTRACTOR STATE MARKER COLOR
    markerShape  = '*';
    attractorStateMarks = scatter3(x(attractorIndicies), ...
        y(attractorIndicies), Z(attractorIndicies) + zShift, 120, markerColor, 'Marker', markerShape,'LineWidth', 2);
    zShift = zShift + infac;
else
    attractorStateMarks = [];
end
if ~isempty(transientStateindiciesOriginal)
    markerColor =  [0.3 1 0.1]; % TRANSIENT STATE MARKER
    markerShape  = '+';
    numOfPoints = numel(transientStateindicies);
    pointLabels = cellfun(@num2str, num2cell(1:numOfPoints),'un',0);
    pointLabels(1) = {'Start (1)'};
    pointLabels(end) = {['End', ' (', num2str(numOfPoints),')']};
    pointLabels = fliplr(pointLabels);
    Length = 3;
    BaseAngle = 30;
    TipAngle = 60;
    Width = 1;
    transientStateMarks = gobjects(numel(transientStateindicies)*2, 1);
    arrowsObj = gobjects(numel(transientStateindicies)-1, 1);
    colorS = jet(numel(transientStateindicies));
    for i = 1:numel(transientStateindicies)
        index = transientStateindicies(i);
        transientStateMarks(i) = scatter3(x(index), ...
            y(index), Z(index) + zShift, 40, markerColor, 'Marker', markerShape,'LineWidth', 2);
        if i == 1 || i == numel(transientStateindicies)
            fontSizE = 15;
        else
            fontSizE = 12;
        end
        transientStateMarks(i + numel(transientStateindicies)) = text(x(index) + shiftBy*1.2, y(index) + shiftBy*1.2, Z(index) + zShift, pointLabels(i), 'Color', [0 0 0], 'FontSize', fontSizE, 'FontWeight', 'bold');
        if i < numel(transientStateindicies)
            startInd = transientStateindicies(i + 1) ;
            Start = [x(startInd), y(startInd), Z(startInd)];
            Stop = [x(index), y(index), Z(index)];
            arrowsObj(i) = arrow(Start, Stop, Length, BaseAngle, TipAngle, Width, 'FaceColor', [0 0 0], 'EdgeColor', colorS(i, :));% [255 140 0.1]./260);
            arrowsObj(i).Visible = 'off';
        end
    end
    zShift = zShift + infac;
else
    arrowsObj = [];
    transientStateMarks = [];
end
if ~isempty(prominentSteadyStateIndiciesOriginal)
    markerColor =  [0.1137 0.4667 0.9216]; % PROMINENT STATE MARKER
    markerShape  = 'x';
    prominentSteadyStateMarks = scatter3(x(prominentSteadyStateIndicies), ...
        y(prominentSteadyStateIndicies), Z(prominentSteadyStateIndicies) + zShift, 80, markerColor, 'Marker', markerShape,'LineWidth', 2);
else
    prominentSteadyStateMarks = [];
end
end

function [gridTipToolData, gridData] = assembleToolTipData(myNetwork, x, y, z, randomInd, stateSpaceLen, mappingMethod, stateSpace, attractorIndiciesOriginal, transientStateindiciesOriginal, prominentSteadyStateIndiciesOriginal)
allOrderInTrajectory = [];
allCellFates = [];
allBasinSizes = [];
allProbabilities = [];
allAttractorTypes = [];
allAttractorNumber = [];
allOutputNodeActivities = [];
allAvgOutputNodeActivities = [];
if ~isempty(attractorIndiciesOriginal)
    [orderInTrajectory, attractorNumber, basinSizes, probabilities, cellFates...
        , attractorTypes, outputNodeActivities...
        , avgOutputNodeActivities] = generate_AttractorStateToolTipData(attractorIndiciesOriginal...
        , transientStateindiciesOriginal, myNetwork, stateSpace);
    allOrderInTrajectory = vertcat(allOrderInTrajectory, orderInTrajectory);
    allAttractorNumber = vertcat(allAttractorNumber, attractorNumber);
    allCellFates = vertcat(allCellFates, cellFates);
    allBasinSizes =  vertcat(allBasinSizes, basinSizes);
    allProbabilities =  vertcat(allProbabilities, probabilities);
    allAttractorTypes =  vertcat(allAttractorTypes, attractorTypes);
    allOutputNodeActivities = vertcat(allOutputNodeActivities, outputNodeActivities);
    allAvgOutputNodeActivities = vertcat(allAvgOutputNodeActivities, avgOutputNodeActivities);
end
if ~isempty(transientStateindiciesOriginal) || ~isempty(prominentSteadyStateIndiciesOriginal)
    [orderInTrajectory, attractorNumber, basinSizes, probabilities, cellFates, attractorTypes, outputNodeActivities...
        , avgOutputNodeActivities] = generate_SteadyStatesToolTipData(transientStateindiciesOriginal...
        , prominentSteadyStateIndiciesOriginal, myNetwork, stateSpace);
    allOrderInTrajectory = vertcat(allOrderInTrajectory, orderInTrajectory);
    allAttractorNumber = vertcat(allAttractorNumber, attractorNumber);
    allCellFates = vertcat(allCellFates, cellFates);
    allBasinSizes =  vertcat(allBasinSizes, basinSizes);
    allProbabilities =  vertcat(allProbabilities, probabilities);
    allAttractorTypes =  vertcat(allAttractorTypes, attractorTypes);
    allOutputNodeActivities = vertcat(allOutputNodeActivities, outputNodeActivities);
    allAvgOutputNodeActivities = vertcat(allAvgOutputNodeActivities, avgOutputNodeActivities);
end
empty = cellfun(@isempty, allCellFates);
allCellFates(empty) = {'Uncharacterized'};
totalStates = int2str(size(stateSpace, 1));
if strcmp(mappingMethod, 'naive')
    allOriginalIndicies = vertcat(attractorIndiciesOriginal, transientStateindiciesOriginal...
        , prominentSteadyStateIndiciesOriginal);
    gridTipToolData = cell(stateSpaceLen, stateSpaceLen);
    if size(x, 2) ~= 1
        gridData = ones(numel(x), 2);
        for i = 1:numel(x)
            gridData(i, :) = [x(i), y(i)];
        end
    else
        gridData = [x, y];
    end

function [randomInd, Z] = groupCyclicAttractors(myNetwork, stateSpace, randomInd, stateSpaceLen, Z)
cancelButton = CancelSimulationButton;
movegui(cancelButton, 'northeast');
setappdata(0,'CancelSimulation', []);
drawnow;
randomIndSquare = reshape(randomInd, [stateSpaceLen, stateSpaceLen]);
cyclicAttractors = cell(size(myNetwork.AttractorTable, 1), 1);
for i = 1:size(myNetwork.AttractorTable, 1)
    attr = myNetwork.AttractorTable{i, 1};
    if size(attr, 1) > 1
        cyclicAttractors(i) = myNetwork.AttractorTable(i, 1);
    end
end
empty = cellfun(@isempty, cyclicAttractors);
if empty == 0
    empty = [];
end
cyclicAttractors(empty) = [];
unavailablePositions = [];
if ~isempty(cyclicAttractors)
    for i = 1:numel(cyclicAttractors)
        drawnow;
        if ~isempty(getappdata(0,'CancelSimulation'))
            cancelButton.delete;
            undefined/0;
        end
        indicieS = randomInd(1:size(stateSpace, 1));
        [~, currStatePositions] = ismember(cyclicAttractors{i}, stateSpace(indicieS, :), 'rows');
        numOfStates = size(cyclicAttractors{i}, 1);
        [r, c] = cellfun(@(currPos) find(randomIndSquare == randomIndSquare(currPos)), num2cell(currStatePositions, 2));
        oldPositions = [r, c];
        [scores, distances, availableSetsOfPoints] = cellfun(@(rc) findValidPositions(rc(:, 1), rc(:, 2), ...
            numOfStates, unavailablePositions, stateSpaceLen), num2cell(oldPositions, 2), 'un', 0);
        [~, centerStateIndex] = min(cell2mat(scores));
        [~, pointIndicies] = sort(distances{centerStateIndex});
        newPositions = availableSetsOfPoints{centerStateIndex}(pointIndicies(1:numOfStates-1), :);
        associatedWeights = distances{centerStateIndex}(pointIndicies(1:numOfStates-1), :).*2;
        unavailablePositions = vertcat(unavailablePositions, oldPositions(centerStateIndex, :), newPositions);
        oldPositions(centerStateIndex, :) = [];
        for j = 1:size(newPositions, 1)
            rNew = newPositions(j, 1); cNew = newPositions(j, 2);
            rOld = oldPositions(j, 1); cOld = oldPositions(j, 2);
            temp1 = randomIndSquare(rNew, cNew);
            temp2 = Z(rNew, cNew);
            randomIndSquare(rNew, cNew) = randomIndSquare(rOld, cOld);
            randomIndSquare(rOld, cOld) = temp1;
            Z(rNew, cNew) = Z(rOld, cOld)/associatedWeights(j);
            Z(rOld, cOld) = temp2;
        end
    end
    randomInd = randomIndSquare(:);
end
cancelButton.delete;
end

function [y, E] = sammonMapping(x, n, opts)
cancelButton = CancelSimulationButton;
movegui(cancelButton, 'northeast');
setappdata(0,'CancelSimulation', []);
progressbar('Projecting State Space Using Sammon Mapping...');
if nargin < 3
    opts.Display        = 'NOiter';
    opts.Input          = 'raw';
    opts.MaxHalves      = 40;
    opts.MaxIter        = 500;
    opts.TolFun         = 1e-9;
    opts.Initialisation = 'pca';
end
if nargin == 0
    y = opts;
    return;
end
if nargin < 2
    n = 2;
end
if strcmp(opts.Display, 'iter')
    display = 2;
elseif strcmp(opts.Display, 'on')
    display = 1;
else
    display = 0;
end
if strcmp(opts.Input, 'distance')
    D = x;
else
    D = euclid(x, x);
end
N     = size(x, 1);
scale = 0.5 / sum(D(:));
D     = D + eye(N);
Dinv  = 1 ./ D;
if strcmp(opts.Initialisation, 'pca')
    [UU,DD] = svd(x);
    y       = UU(:,1:n)*DD(1:n,1:n);
else
    y = randn(N, n);
end
one   = ones(N,n);
d     = euclid(y,y) + eye(N);
dinv  = 1./d;
delta = D - d;
E     = sum(sum((delta.^2).*Dinv));
for i=1:opts.MaxIter
    progressbar(i/opts.MaxIter);
    if ~isempty(getappdata(0,'CancelSimulation'))
        cancelButton.delete;
        undefined/0;
    end
    delta    = dinv - Dinv;
    deltaone = delta * one;
    g        = delta * y - y .* deltaone;
    dinv3    = dinv .^ 3;
    y2       = y .^ 2;
    H        = dinv3 * y2 - deltaone - 2 * y .* (dinv3 * y) + y2 .* (dinv3 * one);
    s        = -g(:) ./ abs(H(:));
    y_old    = y;
    for j=1:opts.MaxHalves
        y(:) = y_old(:) + s;
        d     = euclid(y, y) + eye(N);
        dinv  = 1 ./ d;
        delta = D - d;
        E_new = sum(sum((delta .^ 2) .* Dinv));
        if E_new < E
            break;
        else
            s = 0.5*s;
        end
    end
    if j == opts.MaxHalves
        warning('MaxHalves exceeded. Sammon mapping may not converge...');
    end
    if abs((E - E_new) / E) < opts.TolFun
        if display
            fprintf(1, 'Optimisation terminated - TolFun exceeded.\n');
        end
        progressbar(1);
        break;
    end
    E = E_new;
    if display > 1
        fprintf(1, 'epoch = %d : E = %12.10f\n', i, E * scale);
    end
end
cancelButton.delete;
E = E * scale;
    function d = euclid(x, y)
        d = sqrt(sum(x.^2,2)*ones(1,size(y,1))+ones(size(x,1),1)*sum(y.^2,2)'-2*(x*y'));
    end
end

function [figHandle, bool, plotHandle, gridTipToolData, gridData] = plot_AttractorLandscape(myNetwork, x, y, z, plottingOptions, stateSpace)
try
    plotTitle = plottingOptions.plotTitle;
    showToolTip = plottingOptions.showToolTip;
    annotateAttractors = plottingOptions.AnnotateAttractors;
    selectedColorScheme = plottingOptions.selectedColorScheme;
    mappingMethod = plottingOptions.mappingMethod;
    figure;
    if strcmp(mappingMethod, 'sammon')
        resolution = (max(x(:))-min(x(:)))/(sqrt(size(stateSpace, 1)*4));
        xn = min(x):resolution:max(x);
        yn = min(y):resolution:max(y);
        [xq,yq] = meshgrid(xn, yn);
        stateSpaceLen = length(xq);
        zq = griddata(x, y, z, xq, yq,'natural');
        plotHandle = surfc(xq, yq, zq);
        Z = z;
        x = xq;
        y = yq;
    elseif strcmp(mappingMethod, 'naive')
        stateSpaceLen = length(x);
        if stateSpaceLen^2 ~= size(stateSpace, 1)
            numExtraBox = stateSpaceLen^2 - size(stateSpace, 1);
            extraValues = zeros(numExtraBox, 1);
            extraValues(:) = 0;
            newZ = vertcat(z, extraValues);
        else
            newZ = z;
        end
        if ~isempty(myNetwork.RandomIndicies) && size(myNetwork.RandomIndicies, 1) == size(newZ, 1)
            randomInd = myNetwork.RandomIndicies;
        else
            randomInd = randperm(size(stateSpace, 1))';
            extraVals = size(stateSpace, 1) + 1 : size(newZ, 1);
            randomInd = vertcat(randomInd, extraVals');
            myNetwork.RandomIndicies = randomInd;
        end
        shuffledZ = newZ(randomInd);
        Z = reshape(shuffledZ, [stateSpaceLen, stateSpaceLen]);
        if ~isempty(myNetwork.AttractorTable) && plottingOptions.groupCyclicAttractors
            [randomInd, Z] = groupCyclicAttractors(myNetwork, stateSpace, randomInd, stateSpaceLen, Z);
        end
        plotHandle = surfc(x, y, Z);
        Z = reshape(newZ(randomInd), [stateSpaceLen, stateSpaceLen]);
    end

FILE Toolbox\Visualize Results\Attractor Landscape\Naive Random Projection\Supplementary Functions\findPositions.m
function [x , y] = findPositions(r, c, positionsNeeded, stateSpaceLen)
dx = 1; 
dy = 1;
x = zeros(positionsNeeded, 1);
y = zeros(positionsNeeded, 1);
positionsFound = 2;
count = 1;
while positionsFound <= positionsNeeded
    populateXY(0, dy);
    if positionsFound > positionsNeeded
        break;
    end
    populateXY(dx, 0);
    count = count + 1;
    dx = -dx;
    dy = -dy;
end
x = x + r;
y = y + c;
invalidIndicies = x <= 0;
x(invalidIndicies) = [];
y(invalidIndicies) = [];
invalidIndicies = y <= 0;
x(invalidIndicies) = [];
y(invalidIndicies) = [];
invalidIndicies = x > stateSpaceLen;
x(invalidIndicies) = [];
y(invalidIndicies) = [];
invalidIndicies = y > stateSpaceLen;
x(invalidIndicies) = [];
y(invalidIndicies) = [];
x(1) = [];
y(1) = [];
    function populateXY(dx, dy)
        for j = 1 : count
            x(positionsFound) = x(positionsFound-1) + dx;
            y(positionsFound) = y(positionsFound-1) + dy;
            positionsFound = positionsFound + 1;
            if positionsFound > positionsNeeded
                break;
            end
        end
    end
end

FILE Toolbox\Visualize Results\Attractor Landscape\Naive Random Projection\Supplementary Functions\findValidPositions.m
function [score, distances, availablePoints] = findValidPositions(r, c, pointsNeeded, unavailablePositions, stateSpaceLen)
[rows, cols] = findPositions(r, c, pointsNeeded*2 + numel(unavailablePositions), stateSpaceLen);
if ~isempty(unavailablePositions)
    [~, indicies] = ismember(unavailablePositions, [rows, cols], 'rows');
    indicies(indicies == 0) = [];
    rows(indicies) = [];
    cols(indicies) = [];
end
availablePoints = [rows, cols];
distances = cellfun(@(x) norm([r,c] - x) , num2cell(availablePoints,2));
if  size(distances, 1) >= pointsNeeded
    score = mean(distances) + var(distances);
elseif size(distances, 1) < pointsNeeded
    score = inf;
end
end

function bool = plot_CellFateLandscape(myNetwork, plotTitle, showLegends, annotateBoxes, selectedColorScheme, CellFateLandsacpe)
try

    end
    bool = 1;
catch
    bool = 0;
end
    function rectangles = treemap(data,w,h)
        set(0,'RecursionLimit',5000)
        if (nargin == 0)
            data = abs(randn(1,80)) .^ 2;
        end
        if(nargin < 2)
            w = 1;
            h = 1;
        end
        areaRatio = (w * h) / sum(data);
        [v,sortIndex] = sort(data,'descend');
        p = partitionArea(v,w,h);
        rectangles = zeros(4,length(data));
        oX = 0;
        oY = 0;
        first = 1;
        resultIndex = 1;
        for i = 1:length(p)
            last = first + p(i) - 1;
            chunk = v(first:last);
            first = last + 1;
            blockArea = chunk * areaRatio;
            columnArea = sum(blockArea);
            blockX = oX;
            blockY = oY;
            if((w < h)) % a row
                columnHeight = columnArea / w;
                for j = 1:p(i)
                    blockWidth = blockArea(j) / columnHeight;
                    rectangles(:,sortIndex(resultIndex)) = [blockX,blockY,blockWidth,columnHeight];
                    resultIndex = resultIndex + 1;
                    blockX = blockX + blockWidth;
                end
                oY = oY + columnHeight;
                h = h - columnHeight;
            else % a column
                columnWidth = columnArea / h;
                for j = 1:p(i)
                    blockHeight = blockArea(j) / columnWidth;
                    rectangles(:,sortIndex(resultIndex)) = [blockX,blockY,columnWidth,blockHeight];
                    resultIndex = resultIndex + 1;
                    blockY = blockY  +  blockHeight;
                end
                oX = oX + columnWidth;
                w = w - columnWidth;
            end
        end
        if(nargout == 0)
            plotRectangles(rectangles)
        end
    end
    function partition = partitionArea(v,w,h)
        if(w>h)
            longSide = w;
            shortSide = h;
        else
            longSide = h;
            shortSide = w;
        end
        areaRatio = (w * h) / sum(v);
        bestAverageAspectRatio = inf;
        partition = [];
        for i = 1:length(v)
            columnTotal = sum(v(1:i));
            columnArea = columnTotal * areaRatio;
            columnWidth = columnArea / shortSide;
            sumOfAspectRatio = 0;
            for j = 1:i
                blockArea = v(j) * areaRatio;
                blockHeight = blockArea / columnWidth;
                aspectRatio = blockHeight / columnWidth;
                if(aspectRatio < 1)
                    aspectRatio = 1 / aspectRatio;
                end
                sumOfAspectRatio = sumOfAspectRatio + aspectRatio;
            end
            averageAspectRatio = sumOfAspectRatio/i;
            if(averageAspectRatio >= bestAverageAspectRatio)
                if(partition < i) % if we are not done yet, we recurse
                    p = partitionArea(v(i:end),shortSide,longSide - columnWidth);
                    partition = [partition,p]; %#ok<AGROW>
                end
                return
            end
            bestAverageAspectRatio = averageAspectRatio;
            partition = i;
        end
    end
    function [textObj, patchObj] = plotRectangles(rectangles,labels,colors)
        if(nargin < 2)
            labels = [];
        end
        if(nargin < 3)
            colors = rand(size(rectangles,2),3).^0.5;
        end
        textObj = gobjects(size(rectangles,2), 1);
        patchObj = gobjects(size(rectangles,2), 1);
        for i = 1:size(rectangles,2)
            r = rectangles(:,i);
            xPoints = [r(1), r(1), r(1) + r(3),r(1) + r(3)];
            yPoints = [r(2), r(2)+ r(4), r(2)+ r(4),r(2)];
            patchObj(i) = patch(xPoints, yPoints, colors(i,:),'EdgeColor','none');
            if(~isempty(labels))
                textObj(i) = text(r(1) + r(3)/2,r(2) + r(4)/2, 1, labels{i}, ...
                    'VerticalAlignment','middle','HorizontalAlignment','center');
            end
        end
        axis equal
        axis tight
        axis off
    end
    function outline(rectangles)
        for i = 1:size(rectangles,2)
            r = rectangles(:,i);
            xPoints = [r(1),      r(1), r(1)+r(3), r(1)+r(3)];
            yPoints = [r(2), r(2)+r(4), r(2)+r(4), r(2)     ];
            patch(xPoints,yPoints,[0 0 0],'FaceColor','none')
        end
    end
end
